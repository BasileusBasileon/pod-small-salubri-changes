namespace = POD_salubri_changes

POD_salubri_changes.1 = {
	type = character_event
	content_source = dlc_POD
	title = POD_salubri_changes.1.title
	desc = POD_salubri_changes.1.desc

	theme = salubri_coa
	override_background = { reference = ep2_hunt_foggy_forest }

	left_portrait = {
		character = root
		animation = throne_room_conversation_4
		camera = camera_event_very_left
	}
	right_portrait = {
		character = scope:warrior
		animation = throne_room_conversation_3
	}
	
	widget = {
		gui = "POD_shader_event_fog_alley_night"
		container = "background_shader_vfx_container"
	}
	
	immediate = {
		play_music_cue = "mood_in_search_of_iniquity"

		if = {
			limit = {
				any_character_artifact = {
					has_variable = code_samiel
				}
			}
			random_relation = {
				type = friend
				type = lover
				limit = {
					has_trait = salubri_warrior
				}
				save_scope_as = warrior
			}
		}
		else = {
			random_relation = {
				type = friend
				type = lover
				limit = {
					has_trait = salubri_warrior
					any_character_artifact = {
						has_variable = code_samiel
					}
				}
				save_scope_as = warrior
			}
		}
	}
	
	option = {
		name = POD_salubri_changes.1.option.a
		custom_tooltip = POD_salubri_changes.1.option.a.tt

		if = {
			limit = {
				NOR = {
					has_trait = witch
					any_secret = {
						type = secret_witch
					}
				}
			}

			trigger_event = POD_salubri_changes.2
		}
		else = {
			trigger_event = POD_salubri_changes.3
		}
	}
	
	option = {
		trigger = {
			NOR = {
				has_trait = witch
				any_secret = {
					type = secret_witch
				}
			}
		}

		name = POD_salubri_changes.1.option.b
		
		# Refund
		add_piety = 10000
	}
	
	option = {
		trigger = {
			trigger_if = {
				limit = {
					NOR = {
						has_trait = witch
						any_secret = {
							type = secret_witch
						}
					}
				}
				always = no
			}
		}

		name = POD_salubri_changes.1.option.c
		
		# Refund
		add_piety = 10000
	}
}
